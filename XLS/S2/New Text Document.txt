Function CleanSchoolName(ByVal txt As String) As String
    Dim i As Long
    Dim char As String
    Dim result As String
    Dim forbiddenWords As Variant
    Dim word As Variant
    
    ' قائمة الكلمات الممنوعة
    forbiddenWords = Array("مدرسة", "اعدادية", "ثانوية", "للبنات", "للبنين", "بنين", "بنات", "مختلطة", "المختلطة", "مخ")
    
    ' 1. إزالة النقاط والأرقام والرموز والمدات (يبقى فقط العربي والمسافة)
    txt = Replace(txt, ".", "")
    txt = Replace(txt, "ـ", "")
    txt = Replace(txt, "/", "")
    txt = Replace(txt, "(", "")
    txt = Replace(txt, ")", "")
    ' يمكنك إضافة المزيد من الرموز هنا
    
    ' 2. إزالة الأرقام
    For i = 0 To 9
        txt = Replace(txt, i, "")
    Next i
    
    ' 3. إزالة الكلمات الممنوعة
    For Each word In forbiddenWords
        txt = Replace(txt, word, "")
    Next word
    
    ' 4. إزالة الحروف المفردة (مثل ع ، م ، ث)
    Dim wordsArray() As String
    wordsArray = Split(txt, " ")
    For i = LBound(wordsArray) To UBound(wordsArray)
        If Len(Trim(wordsArray(i))) > 1 Then
            result = result & wordsArray(i) & " "
        End If
    Next i
    
    CleanSchoolName = Trim(result)
End Function
